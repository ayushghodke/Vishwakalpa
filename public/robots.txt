# Crawling is ALLOWED even while the site is behind the maintenance page.
#
# That is deliberate and it is the opposite of most people's instinct. While the
# site is dark it is kept out of the index by an "X-Robots-Tag: noindex" header
# in vercel.json — but a crawler can only see that header if it is allowed to
# fetch the page. Blocking with "Disallow: /" would hide the noindex, and Google
# can still index a URL it has never crawled if it finds a link to it elsewhere.
# Blocking would therefore make the problem it appears to solve harder to fix.

User-agent: *
Allow: /
